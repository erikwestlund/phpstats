Short analyses of the small datasets from phit’s documentation. The output under each example is what it printed when the documentation was built.

Data: schools.csv · housing.csv · learning.csv · clinic.csv

Imports for every example on this page. The model examples also need phit; see Models.

use Phibble\DataFrame;use function Phplyr\{phplyr, mutate, group_by, summarize};use function Phplyr\{arrange, desc, head, pivot_wider};use function Phplyr\{across, where, case_when, if_else};use function Phplyr\{n, mean, min, max, ntile, min_rank};use function Phplyr\{lm, glm, lmer};

School residuals

schools.csv has 250 students in 12 schools, with study hours, a prior score, and a final score.

  • PHP 8.5+
  • PHP 8.3–8.4
// Fit one regression on all students, then average// each school's residuals.$schools = DataFrame::fromCsv('schools.csv');$fit = $schools |>    lm('score ~ prior_score + study_hours');[$b0, $bPrior, $bHours] = $fit['coefficients'];$expected = fn ($r) => $b0    + $bPrior * $r->prior_score    + $bHours * $r->study_hours;$shareAbove = fn ($g) => round(\count(array_filter(    $g->column('residual'),    fn ($e) => $e > 0,)) / $g->nrow(), 2);echo $schools |>    mutate(        residual: fn ($r) => $r->score - $expected($r),    ) |>    group_by('school') |>    summarize(        students: n(),        avg_score: mean('score'),        avg_residual: mean('residual'),        share_above: $shareAbove,    ) |>    mutate(        rank: min_rank(desc('avg_residual')),        avg_score: fn ($r) => round($r->avg_score, 1),        avg_residual: fn ($r) => round($r->avg_residual, 2),    ) |>    arrange('rank');
// Fit one regression on all students, then average// each school's residuals.$schools = DataFrame::fromCsv('schools.csv');$fit = phplyr($schools)    ->lm('score ~ prior_score + study_hours');[$b0, $bPrior, $bHours] = $fit['coefficients'];$expected = fn ($r) => $b0    + $bPrior * $r->prior_score    + $bHours * $r->study_hours;$shareAbove = fn ($g) => round(\count(array_filter(    $g->column('residual'),    fn ($e) => $e > 0,)) / $g->nrow(), 2);echo phplyr($schools)    ->mutate(        residual: fn ($r) => $r->score - $expected($r),    )    ->groupBy('school')    ->summarize(        students: n(),        avg_score: mean('score'),        avg_residual: mean('residual'),        share_above: $shareAbove,    )    ->mutate(        rank: min_rank(desc('avg_residual')),        avg_score: fn ($r) => round($r->avg_score, 1),        avg_residual: fn ($r) => round($r->avg_residual, 2),    )    ->arrange('rank');

Schools as random intercepts

  • PHP 8.5+
  • PHP 8.3–8.4
// Students nested in schools: shared fixed effects and// one intercept shift per school.$schools = DataFrame::fromCsv('schools.csv');$formula = 'score ~ study_hours + prior_score'    . ' + (1 | school)';$fit = $schools |> lmer($formula);foreach ($fit['columns'] as $i => $term) {    printf(        "%-12s %8.4f  (se %.4f)\n",        $term,        $fit['coefficients'][$i],        $fit['std_errors'][$i],    );}$schoolSd = $fit['theta'][0] * $fit['sigma'];$icc = $schoolSd ** 2    / ($schoolSd ** 2 + $fit['sigma'] ** 2);printf(    "school sd %.3f, residual sd %.3f, ICC %.3f\n\n",    $schoolSd,    $fit['sigma'],    $icc,);// Each school's raw mean score beside its estimated// intercept shift.$random = $fit['random'][0];$shifts = array_combine(    $random['group_labels'],    array_column($random['effects'], 0),);echo $schools |>    group_by('school') |>    summarize(students: n(), raw_mean: mean('score')) |>    mutate(        intercept_shift: fn ($r) =>            round($shifts[$r->school], 2),        raw_mean: fn ($r) => round($r->raw_mean, 1),    ) |>    arrange(desc('intercept_shift')) |>    head(5);

Rent per square meter

housing.csv has eight flats.

  • PHP 8.5+
  • PHP 8.3–8.4
// Rent per square meter by number of bedrooms, and each// flat's difference from its bedroom group's mean.$housing = DataFrame::fromCsv('housing.csv') |>    mutate(per_m2: fn ($r) => round(        $r->monthly_rent / $r->area_m2,        2,    ));echo $housing |>    group_by('bedrooms') |>    summarize(        across(            ['area_m2', 'monthly_rent', 'per_m2'],            ['mean' => mean(...), 'max' => max(...)],        ),        flats: n(),    ) |>    mutate(across(        where('double'),        fn ($v) => round($v, 1),    ));$vsGroup = fn ($r, $g) => round(    $r->per_m2 - mean('per_m2')($g),    2,);echo "\n", $housing |>    mutate(vs_group: $vsGroup, _by: 'bedrooms') |>    arrange('bedrooms', desc('vs_group'));

Pass rates by study time

learning.csv has 80 students, with study hours, a practice score, and whether each passed.

  • PHP 8.5+
  • PHP 8.3–8.4
// Pass rates by study-hour quartile, then a logistic// regression.$learning = DataFrame::fromCsv('learning.csv');echo $learning |>    mutate(hours_quartile: ntile('study_hours', 4)) |>    group_by('hours_quartile') |>    summarize(        students: n(),        min_hours: min('study_hours'),        max_hours: max('study_hours'),        pass_rate: mean('passed'),    );$fit = $learning |> glm(    'passed ~ study_hours + practice_score',    family: 'binomial',);printf(    "\nOdds ratio for study_hours: %.3f\n",    exp($fit['coefficients'][1]),);
// Pass rates by study-hour quartile, then a logistic// regression.$learning = DataFrame::fromCsv('learning.csv');echo phplyr($learning)    ->mutate(hours_quartile: ntile('study_hours', 4))    ->groupBy('hours_quartile')    ->summarize(        students: n(),        min_hours: min('study_hours'),        max_hours: max('study_hours'),        pass_rate: mean('passed'),    );$fit = phplyr($learning)->glm(    'passed ~ study_hours + practice_score',    family: 'binomial',);printf(    "\nOdds ratio for study_hours: %.3f\n",    exp($fit['coefficients'][1]),);

Clinic calls by staffing

clinic.csv has visitors, staff, and calls for 120 days.

  • PHP 8.5+
  • PHP 8.3–8.4
// Mean calls by staffing level and visitor volume, as a// cross-tabulation.$levels = ['low', 'mid', 'high'];echo DataFrame::fromCsv('clinic.csv') |>    mutate(        staffing: case_when(            [fn ($r) => $r->staff < 4, 'low'],            [fn ($r) => $r->staff < 6, 'mid'],            default: 'high',        ),        volume: if_else(            fn ($r) => $r->visitors >= 250,            'busy',            'quiet',        ),    ) |>    group_by('staffing', 'volume') |>    summarize(mean_calls: mean('calls'), _groups: 'drop') |>    mutate(        mean_calls: fn ($r) => round($r->mean_calls, 2),    ) |>    pivot_wider(        namesFrom: 'volume',        valuesFrom: 'mean_calls',        namesSort: true,    ) |>    arrange(        fn ($r) => array_search($r->staffing, $levels),    );
// Mean calls by staffing level and visitor volume, as a// cross-tabulation.$levels = ['low', 'mid', 'high'];echo phplyr(DataFrame::fromCsv('clinic.csv'))    ->mutate(        staffing: case_when(            [fn ($r) => $r->staff < 4, 'low'],            [fn ($r) => $r->staff < 6, 'mid'],            default: 'high',        ),        volume: if_else(            fn ($r) => $r->visitors >= 250,            'busy',            'quiet',        ),    )    ->groupBy('staffing', 'volume')    ->summarize(mean_calls: mean('calls'), _groups: 'drop')    ->mutate(        mean_calls: fn ($r) => round($r->mean_calls, 2),    )    ->pivotWider(        namesFrom: 'volume',        valuesFrom: 'mean_calls',        namesSort: true,    )    ->arrange(        fn ($r) => array_search($r->staffing, $levels),    );
